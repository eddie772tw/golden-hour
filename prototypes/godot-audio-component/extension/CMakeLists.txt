cmake_minimum_required(VERSION 3.20)
project(gh_audio LANGUAGES CXX)
set(GODOT_CPP_DIR "" CACHE PATH "Path to the pinned godot-cpp source checkout")
if(NOT EXISTS "${GODOT_CPP_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Provide -DGODOT_CPP_DIR=<godot-cpp checkout>")
endif()
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/profile.json" CACHE PATH "")
add_subdirectory("${GODOT_CPP_DIR}" godot-cpp)
add_library(gh_audio SHARED playback.cpp)
target_compile_features(gh_audio PRIVATE cxx_std_20)
target_link_libraries(gh_audio PRIVATE godot-cpp ws2_32)
if(NOT MSVC)
    target_compile_options(gh_audio PRIVATE -ffp-contract=off)
endif()
if(MINGW)
    target_link_options(gh_audio PRIVATE -static)
endif()
set(probe_project "${CMAKE_BINARY_DIR}/project")
foreach(script IN ITEMS project.godot audio.gdextension ipc.gd scene_audio.gd)
    configure_file("project/${script}" "${probe_project}/${script}" COPYONLY)
endforeach()
set_target_properties(gh_audio PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${probe_project}/bin")
